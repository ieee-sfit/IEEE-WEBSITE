    const buildArm = (isLeft: boolean) => {
      const armRoot = new THREE.Object3D();
      skeleton.add(armRoot);

      // 1. Spacing: Hands are wide, slightly below core
      const pX = isLeft ? -30 : 30; 
      const pY = -5;
      const pZ = 12;
      
      const palm = new THREE.Object3D();
      palm.position.set(pX, pY, pZ);
      
      // 2. True Anatomical Orientation
      // Calculate a target directly AWAY from the core.
      // Core is at (0,0,0). Vector from Core to Palm is exactly palm.position.
      // So looking at palm.position * 2 points local +Z directly AWAY from core into the void.
      // This intrinsically points local -Z (the front of the hand) directly at the core!
      const awayTarget = palm.position.clone().multiplyScalar(2);
      palm.lookAt(awayTarget);
      
      // Tilt the palm slightly so it cups from below instead of hitting head-on
      palm.rotateX(-0.25);
      
      armRoot.add(palm);

      // 3. Intricate Palm Structure (wider for spread fingers)
      const palmPlate1 = new THREE.Object3D();
      palmPlate1.userData = { type: 'carapace', scale: new THREE.Vector3(4.5, 4.5, 0.6) };
      palm.add(palmPlate1);
      
      const palmPlate2 = new THREE.Object3D();
      palmPlate2.position.set(0, 0, 1.5);
      palmPlate2.rotation.z = 0.2;
      palmPlate2.userData = { type: 'carapace', scale: new THREE.Vector3(3.5, 3.5, 0.8) };
      palm.add(palmPlate2);

      // 4. The Wrist
      const wrist = new THREE.Object3D();
      wrist.position.set(0, 0, 6); // +Z is AWAY from core
      palm.add(wrist);

      // 5. The Elaborate Forearm (Extending from Wrist to Elbow)
      const elbow = new THREE.Object3D();
      elbow.position.set(0, 0, 45); // Forearm is 45 units long!
      // Angle the elbow so the upper arm reaches into the sky/void
      elbow.rotation.x = -0.4; // Bend upwards
      elbow.rotation.y = isLeft ? 0.3 : -0.3; // Splay outwards
      wrist.add(elbow);

      // Fill the forearm with structural shards
      for(let i=0; i<9; i++) {
         const zPos = 2 + i * 5;
         const fShard = new THREE.Object3D();
         fShard.position.set((Math.random() - 0.5)*3, (Math.random() - 0.5)*3, zPos);
         fShard.rotation.z = (Math.random() - 0.5);
         const s = 1.0 + (i * 0.15); // tapers from elbow down to wrist
         fShard.userData = { type: 'carapace', scale: new THREE.Vector3(s*1.8, s*2.2, 1.5) };
         wrist.add(fShard);

         if (i % 2 === 0) {
            const fArch = new THREE.Object3D();
            fArch.position.set(0, 0, zPos);
            fArch.rotation.x = Math.PI/2;
            fArch.rotation.z = Math.random();
            fArch.userData = { type: 'arch', scale: new THREE.Vector3(s*1.6, s*1.6, 0.8) };
            wrist.add(fArch);
         }
      }

      // 6. The Upper Arm (Stretching from elbow deep into the void)
      for(let i=0; i<15; i++) {
         const zPos = i * 18; // Massive stretch (up to 270 units off screen!)
         const uShard = new THREE.Object3D();
         uShard.position.set((Math.random() - 0.5)*5, (Math.random() - 0.5)*5, zPos);
         uShard.rotation.z = Math.random() * Math.PI;
         uShard.rotation.x = (Math.random() - 0.5) * 0.2;
         
         const s = 2.5 + (i * 0.4); // Gets huge as it goes into the distance
         uShard.userData = { type: 'carapace', scale: new THREE.Vector3(s*2.5, s*2.5, 3.5) };
         elbow.add(uShard);
      }

      // 7. The Fingers (Massively spread across the wide palm, cupping inward along -Z)
      const fingers = isLeft ? [
        // Left hand: +X is INTO screen. -X is OUT of screen (towards camera).
        // Thumb should be OUT of screen (-X) and somewhat low.
        { name: 'Thumb', offset: new THREE.Vector3(-8, 1, 2),  rotX: 0.1,  rotY: -0.7, length: 9, joints: 3, curl: -0.4, scale: 1.6 },
        { name: 'Index', offset: new THREE.Vector3(-4, 6, 0),  rotX: -0.1, rotY: -0.3, length: 11, joints: 4, curl: -0.35, scale: 1.3 },
        { name: 'Middle', offset: new THREE.Vector3(0, 7, -1), rotX: -0.2, rotY: 0.0,  length: 12, joints: 4, curl: -0.38, scale: 1.4 },
        { name: 'Ring', offset: new THREE.Vector3(4, 6, 0),    rotX: -0.1, rotY: 0.3,  length: 11, joints: 4, curl: -0.35, scale: 1.2 },
        { name: 'Pinky', offset: new THREE.Vector3(8, 3, 1),   rotX: 0.0,  rotY: 0.6,  length: 8, joints: 3, curl: -0.4, scale: 1.0 },
      ] : [
        // Right hand: +X is OUT of screen. -X is INTO screen.
        // Thumb should be OUT of screen (+X).
        { name: 'Thumb', offset: new THREE.Vector3(8, 1, 2),   rotX: 0.1,  rotY: 0.7,  length: 9, joints: 3, curl: -0.4, scale: 1.6 },
        { name: 'Index', offset: new THREE.Vector3(4, 6, 0),   rotX: -0.1, rotY: 0.3,  length: 11, joints: 4, curl: -0.35, scale: 1.3 },
        { name: 'Middle', offset: new THREE.Vector3(0, 7, -1), rotX: -0.2, rotY: 0.0,  length: 12, joints: 4, curl: -0.38, scale: 1.4 },
        { name: 'Ring', offset: new THREE.Vector3(-4, 6, 0),   rotX: -0.1, rotY: -0.3, length: 11, joints: 4, curl: -0.35, scale: 1.2 },
        { name: 'Pinky', offset: new THREE.Vector3(-8, 3, 1),  rotX: 0.0,  rotY: -0.6, length: 8, joints: 3, curl: -0.4, scale: 1.0 },
      ];

      fingers.forEach(fd => {
        const fRoot = new THREE.Object3D();
        fRoot.position.copy(fd.offset);
        fRoot.rotation.set(fd.rotX, fd.rotY, 0);
        palm.add(fRoot);

        let currentJoint = fRoot;
        for (let j = 0; j <= fd.joints; j++) {
          const jointNode = new THREE.Object3D();
          if (j > 0) jointNode.position.set(0, 0, -fd.length);
          
          jointNode.rotation.set(fd.curl, 0, 0);
          currentJoint.add(jointNode);
          
          const knuckle = new THREE.Object3D();
          const kScale = fd.scale * (1.2 - (j * 0.2));
          knuckle.userData = { type: 'joint', scale: new THREE.Vector3(kScale, kScale, kScale) };
          jointNode.add(knuckle);

          if (j < fd.joints) {
            const bone = new THREE.Object3D();
            bone.position.set(0, 0, -fd.length / 2);
            bone.rotation.x = -Math.PI / 2;
            bone.rotation.y = Math.PI / 4;
            
            const thickness = 0.55 * kScale;
            const bLen = (fd.length / 14) * 1.15; 
            bone.userData = { type: 'spike', scale: new THREE.Vector3(thickness, bLen, thickness) };
            
            jointNode.add(bone);
          }
          currentJoint = jointNode;
        }
      });
    };
